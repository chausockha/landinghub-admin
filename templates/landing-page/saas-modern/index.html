import { workflow, edgeStep } from "@velox/sdk";

export const reconcileUsage = workflow({
  id: "reconcile-tenant-usage",
  regions: ["iad", "fra", "hnd"],
  retryPolicy: { maxAttempts: 5, backoff: "exponential" },

  async run({ event, step }) {
    // 1. Durable checkpointed aggregation
    const rollup = await step.run("aggregate-meters", () =>
      edgeStep.queryClickhouse({ tenantId: event.data.tenantId })
    );

    // 2. Cryptographically signed ledger commit
    return await step.commit("finalize-invoice", {
      totalEvents: rollup.count,
      p99LatencyMs: 0.74,
    });
  }
});
